০৮ লিস্ট (List)

লিস্ট হলো অর্ডারড, মিউটেবল (পরিবর্তনযোগ্য) এবং ডুপ্লিকেট ভ্যালু রাখা যায় এমন একটি কালেকশন।

li1 = []                       # খালি লিস্ট
li2 = [4, 5, 6]
li3 = list((1, 2, 3))          # টাপল থেকে লিস্ট
li4 = list(range(1, 11))       # => [1, 2, ..., 10]

অ্যাক্সেস, ইনডেক্সিং ও স্লাইসিং

a = ['spam', 'egg', 'bacon', 'tomato', 'ham', 'lobster']
print(a[2:5])    # ['bacon', 'tomato', 'ham']
print(a[::2])    # প্রতি ২ পদক্ষেপে
print(a[::-1])   # রিভার্স লিস্ট

লিস্টের মেথডসমূহ

মেথডকাজ
.append(x)শেষে একটি আইটেম যোগ করে
.extend(iter)আরেকটি iterable-এর সব আইটেম শেষে যোগ করে
.insert(i, x)নির্দিষ্ট ইনডেক্সে আইটেম বসায়
.remove(x)প্রথম যে x পাবে সেটা মুছে
.pop(i)নির্দিষ্ট ইনডেক্সের আইটেম মুছে ও রিটার্ন করে
.sort() / .reverse()ইন-প্লেস সর্ট / উল্টানো
.index(x) / .count(x)প্রথম ইনডেক্স / কতবার আছে
.copy()শ্যালো কপি বানায়
li = [3, 1, 3, 2, 5]
li.sort()
print(li)              # [1, 2, 3, 3, 5]

odd = [1, 3, 5]
odd.extend([9, 11, 13])   # [1, 3, 5, 9, 11, 13]

sorted() বনাম .sort()

li = [3, 1, 2]
new_li = sorted(li)   # নতুন সর্টেড লিস্ট রিটার্ন করে, মূল li অপরিবর্তিত থাকে
li.sort()              # মূল লিস্টকেই সর্ট করে ফেলে (in-place), None রিটার্ন করে

words = ["banana", "kiwi", "apple"]
words.sort(key=len)    # দৈর্ঘ্য অনুসারে সর্ট

ফিল্টার ও কম্প্রিহেনশন দিয়ে লিস্ট জেনারেট

print(list(filter(lambda x: x % 2 == 1, range(1, 20))))
print([x ** 2 for x in range(1, 11) if x % 2 == 1])   # [1, 9, 25, 49, 81]

নেস্টেড লিস্ট (Matrix)

matrix = [[1, 2, 3], [4, 5, 6], [7, 8, 9]]
print(matrix[1][2])   # 6

grid = [[0] * 3 for _ in range(3)]   # সঠিক উপায়ে 2D লিস্ট তৈরি

শ্যালো বনাম ডিপ কপি

import copy
a = [1, [2, 3]]
b = a                    # একই অবজেক্ট, b বদলালে a-ও বদলায়
c = a.copy()              # শ্যালো কপি — ভেতরের লিস্ট এখনও শেয়ার্ড
d = copy.deepcopy(a)      # ডিপ কপি — সবকিছু সম্পূর্ণ আলাদা
