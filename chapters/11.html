১১ ডিকশনারি (Dictionary)

ডিকশনারি হলো key-value পেয়ারের কালেকশন, JSON অবজেক্টের মতো। Python 3.7+ থেকে ডিকশনারি ইনসার্শন-অর্ডার মেনে চলে।

a = {"one": 1, "two": 2, "three": 3}
print(a["one"])       # 1
print(a.keys())        # dict_keys(['one', 'two', 'three'])
print(a.values())       # dict_values([1, 2, 3])
print(a.items())        # dict_items([...])
a.update({"four": 4})   # নতুন key-value যোগ/আপডেট

ডিকশনারির মেথড

মেথডকাজ
.get(key, default)key না থাকলে Error না দিয়ে default রিটার্ন করে
.pop(key)key মুছে ভ্যালু রিটার্ন করে
.setdefault(key, default)key থাকলে ভ্যালু দেয়, না থাকলে সেট করে দেয়
.update(dict2)আরেকটি ডিকশনারির সাথে মার্জ করে
a = {"one": 1, "two": 2}
print(a.get("three", "নাই"))   # 'নাই' — KeyError হবে না

for key, value in a.items():
    print(key, value)

# Python 3.9+: দুটি ডিকশনারি মার্জ করার সহজ পদ্ধতি
merged = {"a": 1} | {"b": 2}   # {'a': 1, 'b': 2}

squares = {x: x**2 for x in range(6)}
