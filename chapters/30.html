৩০ তারিখ ও সময় (datetime)

from datetime import datetime, date, timedelta
now = datetime.now()
today = date.today()

print(now.strftime("%Y-%m-%d %H:%M:%S"))   # '2026-09-26 14:30:00'
parsed = datetime.strptime("19-09-2026", "%d-%m-%Y")

tomorrow = today + timedelta(days=1)
diff = datetime(2026, 12, 31) - datetime(2026, 1, 1)
print(diff.days)
